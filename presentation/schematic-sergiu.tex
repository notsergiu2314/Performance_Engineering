% Schematics of the byte reductions in Sergiu's files (advec_cell, reset/revert, PdV).
% Needs tikz (loaded by beamerthemeRUG) and the colours rugred, rugdark, ruggrey, ruglight.
% Vocabulary: DRAM box left, core box right, arrows = memory traffic (red = traffic this change removes).
% Every picture is 14.8 cm wide: before | after, each panel 7 cm. Use \resizebox{\textwidth}{!}{...}.
\usetikzlibrary{arrows.meta}
\tikzset{
  sdram/.style={rounded corners=2pt, fill=ruglight, draw=ruggrey, line width=0.5pt},
  score/.style={rounded corners=2pt, fill=white, draw=rugdark, line width=0.7pt},
  sbar/.style={fill=ruggrey!35, draw=ruggrey, minimum height=0.4cm, minimum width=3.2cm, inner sep=0pt, font=\scriptsize},
  sbarr/.style={sbar, fill=rugred!18, draw=rugred},
  sgone/.style={sbar, dashed, fill=white, draw=ruggrey!70, text=ruggrey},
  sreg/.style={fill=rugred!18, draw=rugred, minimum height=0.4cm, minimum width=2.0cm, inner sep=0pt, font=\scriptsize},
  skeep/.style={-{Stealth[length=5pt,width=4pt]}, draw=ruggrey, line width=0.7pt},
  scut/.style={-{Stealth[length=5pt,width=4pt]}, draw=rugred, line width=0.7pt},
  stag/.style={draw=rugdark, fill=white, rounded corners=2pt, minimum height=0.5cm, minimum width=1.9cm, inner sep=1pt, font=\small\ttfamily},
  sbuf/.style={draw=ruggrey, fill=ruggrey!35, minimum height=0.6cm, minimum width=3.2cm, inner sep=1pt, font=\scriptsize},
  stitle/.style={font=\small\bfseries},
  slab/.style={font=\scriptsize, text=ruggrey},
  slabr/.style={font=\scriptsize, text=rugred},
  scount/.style={font=\scriptsize, align=center, text width=7cm},
}

% one panel frame: DRAM box and core box, y from #1 to #2
\newcommand{\sframe}[2]{%
  \draw[sdram] (0,#1) rectangle (3.4,#2);
  \draw[score] (4.8,#1) rectangle (7.0,#2);
  \node[slab,anchor=north west] at (0.05,#2-0.02) {DRAM};
  \node[slab,anchor=north] at (5.9,#2-0.02) {core};}

% ---------------------------------------------------------------- (3) dead store, (7) PdV scalar
% one row: #1 y position, #2 x shift, #3 array name, #4 mode (1 write only, 2 store+reload, 3 gone, 4 scalar), #5 scalar name
\newcommand{\schemRow}[5]{%
  \begin{scope}[shift={(#2,#1)}]
    \sframe{-0.65}{0.65}
    \ifcase#4\or
      \node[sbarr] at (1.7,0) {\texttt{#3}};
      \draw[scut] (4.8,0.1) -- (3.3,0.1);
      \draw[scut] (3.3,-0.1) -- (4.8,-0.1);
      \node[slabr] at (4.05,0.38) {store $\cdot$ RFO};
      \node[slabr] at (4.05,-0.38) {never read};
    \or
      \node[sbarr] at (1.7,0) {\texttt{#3}};
      \draw[scut] (4.8,0.14) -- (3.3,0.14);
      \draw[scut] (3.3,0) -- (4.8,0);
      \draw[scut] (3.3,-0.14) -- (4.8,-0.14);
      \node[slabr,font=\tiny] at (4.05,0.4) {store $\cdot$ RFO $\cdot$ reload};
    \or
      \node[sgone] at (1.7,0) {\texttt{#3}};
      \node[slab] at (4.05,0) {no traffic};
    \or
      \node[sgone] at (1.7,0) {\texttt{#3}};
      \node[sreg] at (5.9,0) {\texttt{\tiny #5}};
      \node[slab] at (4.05,0) {no traffic};
    \fi
  \end{scope}}
